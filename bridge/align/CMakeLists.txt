# 깊이 → 컬러 정렬 라이브러리(fast_align)와 시험·벤치, RealSense 도구.
#   Windows:  cmake -S bridge/align -B bridge/align/build -G "Visual Studio 17 2022" -A x64 -DREALSENSE_ROOT=C:/librealsense2/v2.59.1
#             cmake --build bridge/align/build --config Release
#   Linux/Pi: cmake -S bridge/align -B build -G Ninja -DCMAKE_BUILD_TYPE=Release && ninja -C build
#             (librealsense 가 설치돼 있으면 find_package(realsense2) 로 도구까지)
#   aarch64 크로스: -DCMAKE_TOOLCHAIN_FILE=bridge/align/cmake/aarch64-linux-gnu.cmake
cmake_minimum_required(VERSION 3.16)
project(fast_align CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release)
endif()

option(FA_NATIVE "이 기계에 맞춘 명령어(-march=native)" OFF)
option(FA_TOOLS "RealSense 도구 빌드(librealsense 필요)" ON)
set(REALSENSE_ROOT "" CACHE PATH "librealsense SDK 폴더(include/, lib/, bin/)")

if(MSVC)
  # /fp:precise 는 곱·더하기를 합치지 않음(SDK 와 같은 반올림)
  add_compile_options(/utf-8 /W3 /fp:precise /EHsc /D_CRT_SECURE_NO_WARNINGS)
  if(CMAKE_CXX_COMPILER_ID STREQUAL "Clang")
    add_compile_options(/clang:-ffp-contract=off /clang:-O3)
  endif()
  add_compile_options($<$<CONFIG:Release>:/O2>)
else()
  # SDK(librealsense unix_config.cmake)와 같이 -ffp-contract=off: fma 는 NEON 경로를 따라 코드에 직접 쓴 곳만
  add_compile_options(-Wall -Wextra -Wno-unused-parameter -ffp-contract=off)
  add_compile_options($<$<CONFIG:Release>:-O3>)
  if(FA_NATIVE)
    add_compile_options(-march=native)
  endif()
endif()

add_library(fast_align STATIC src/fast_align.cpp)
target_include_directories(fast_align PUBLIC include)
find_package(Threads REQUIRED)
target_link_libraries(fast_align PUBLIC Threads::Threads)

# 시험·벤치(옮겨 온 SDK SIMD 경로가 x86 에서 SSSE3 를 씀)
set(FA_REF_FLAGS "")
if(NOT MSVC AND CMAKE_SYSTEM_PROCESSOR MATCHES "x86_64|AMD64|amd64")
  set(FA_REF_FLAGS -mssse3)
endif()
add_executable(test_align tests/test_align.cpp)
target_link_libraries(test_align PRIVATE fast_align)
target_compile_options(test_align PRIVATE ${FA_REF_FLAGS})
add_executable(bench_align bench/bench_align.cpp)
target_link_libraries(bench_align PRIVATE fast_align)
target_include_directories(bench_align PRIVATE tests)
target_compile_options(bench_align PRIVATE ${FA_REF_FLAGS})
enable_testing()
add_test(NAME test_align COMMAND test_align)

# ---- RealSense 도구 ----
if(FA_TOOLS)
  set(RS_FOUND OFF)
  if(REALSENSE_ROOT)
    find_path(RS_INCLUDE librealsense2/rs.hpp PATHS ${REALSENSE_ROOT}/include NO_DEFAULT_PATH)
    find_library(RS_LIB NAMES realsense2 PATHS ${REALSENSE_ROOT}/lib ${REALSENSE_ROOT}/lib/x64 NO_DEFAULT_PATH)
    if(RS_INCLUDE AND RS_LIB)
      add_library(realsense2_imp UNKNOWN IMPORTED)
      set_target_properties(realsense2_imp PROPERTIES IMPORTED_LOCATION ${RS_LIB} INTERFACE_INCLUDE_DIRECTORIES ${RS_INCLUDE})
      set(RS_TARGET realsense2_imp)
      set(RS_FOUND ON)
    endif()
  else()
    find_package(realsense2 CONFIG QUIET)
    if(realsense2_FOUND)
      set(RS_TARGET realsense2::realsense2)
      set(RS_FOUND ON)
    endif()
  endif()
  if(RS_FOUND)
    add_library(lodepng STATIC third_party/lodepng/lodepng.cpp)
    target_include_directories(lodepng PUBLIC third_party/lodepng)
    foreach(t rs_info rs_rec rs_align_check align_sdk_check)
      add_executable(${t} tools/${t}.cpp)
      target_include_directories(${t} PRIVATE tools tests)
      target_link_libraries(${t} PRIVATE fast_align lodepng ${RS_TARGET})
      target_compile_options(${t} PRIVATE ${FA_REF_FLAGS})
    endforeach()
    if(WIN32 AND REALSENSE_ROOT)
      # 실행 파일 옆에 realsense2.dll
      foreach(t rs_info rs_rec rs_align_check align_sdk_check)
        add_custom_command(TARGET ${t} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
                           ${REALSENSE_ROOT}/bin/realsense2.dll $<TARGET_FILE_DIR:${t}>)
      endforeach()
    endif()
    message(STATUS "RealSense 도구: librealsense ${RS_TARGET}")
  else()
    message(STATUS "RealSense 도구: librealsense 없음 → 라이브러리·시험·벤치만")
  endif()
endif()
